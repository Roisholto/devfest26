package agent

import (
	"context"

	"devfest26/internal/domain"
)

// AddBeneficiary validates and saves supplied beneficiary details using the
// Activity's stable operation key. The service and PostgreSQL enforce their
// boundaries independently; payment approval remains a separate action.
func (r *ToolDispatcher) AddBeneficiary(ctx context.Context, key string, in domain.AddBeneficiaryRequest) (map[string]any, error) {
	if err := in.Validate(); err != nil {
		return nil, err
	}

	var out domain.Beneficiary

	if err := r.Client.Do(ctx, "POST", "/beneficiaries", key, in, &out); err != nil {
		return nil, err
	}

	return toMap(out), nil
}
